set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(dbox246_render_tests Dbox246RenderTests.cpp)

add_test(NAME dbox246_render COMMAND dbox246_render_tests "${CMAKE_CURRENT_SOURCE_DIR}/vectors")

 add_executable(dbox246_integration_tests ../../Tests/Dbox246IntegrationTests.cpp)
 target_include_directories(dbox246_integration_tests PRIVATE ../../Source)
 add_test(NAME dbox246_integration COMMAND dbox246_integration_tests)
endif()
