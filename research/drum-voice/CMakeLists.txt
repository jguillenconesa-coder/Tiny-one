cmake_minimum_required(VERSION 3.22)
project(PocketDrumVoice246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(drum_voice246_tests DrumVoice246Tests.cpp)
target_include_directories(drum_voice246_tests PRIVATE ../../Source)
add_test(NAME drum_voice246 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RunDrumVoice246Tests.py" "$<TARGET_FILE:drum_voice246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/vectors")
