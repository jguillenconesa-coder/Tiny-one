cmake_minimum_required(VERSION 3.22)
project(PocketBend246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(bend246_tests ../../Tests/Bend246Tests.cpp)
target_include_directories(bend246_tests PRIVATE ../../Source)
if(BUILD_TESTING)
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_test(NAME bend246_original COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../Tests/RunBend246Tests.py" "$<TARGET_FILE:bend246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/../../Tests/ReferenceVectors")
endif()
add_executable(bend246_integration_tests ../../Tests/Bend246IntegrationTests.cpp)
target_include_directories(bend246_integration_tests PRIVATE ../../Source)
add_test(NAME bend246_integration COMMAND bend246_integration_tests)
