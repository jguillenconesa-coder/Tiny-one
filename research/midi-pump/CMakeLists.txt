cmake_minimum_required(VERSION 3.22)
project(PocketMidiPump246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(midi_pump246_tests MidiPump246Tests.cpp)
target_include_directories(midi_pump246_tests PRIVATE ../../Source)
add_test(NAME midi_pump246 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RunMidiPump246Tests.py" "$<TARGET_FILE:midi_pump246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/vectors")
