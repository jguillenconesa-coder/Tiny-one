cmake_minimum_required(VERSION 3.22)
project(PocketCoreModulatorFactory246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(core_modulator_factory246_tests CoreModulatorFactory246Tests.cpp)
target_include_directories(core_modulator_factory246_tests PRIVATE ../../Source)
add_test(NAME core_modulator_factory246 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RunCoreModulatorFactory246Tests.py" "$<TARGET_FILE:core_modulator_factory246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/vectors")
add_test(NAME core_modulator_factory_all246 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RunCoreModulatorFactory246Tests.py" "$<TARGET_FILE:core_modulator_factory246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/vectors-all" --all)
