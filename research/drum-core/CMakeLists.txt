cmake_minimum_required(VERSION 3.22)
project(PocketDrumCore246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(drum_core246_tests DrumCore246Tests.cpp)
target_include_directories(drum_core246_tests PRIVATE ../../Source)
add_test(NAME drum_core246 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RunDrumCore246Tests.py" "$<TARGET_FILE:drum_core246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/vectors")
