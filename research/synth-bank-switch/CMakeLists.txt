cmake_minimum_required(VERSION 3.22)
project(PocketSynthBankSwitch246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(synth_bank_switch246_tests SynthBankSwitch246Tests.cpp)
target_include_directories(synth_bank_switch246_tests PRIVATE ../../Source)
add_test(NAME synth_bank_switch246 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RunSynthBankSwitch246Tests.py" "$<TARGET_FILE:synth_bank_switch246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/vectors")
