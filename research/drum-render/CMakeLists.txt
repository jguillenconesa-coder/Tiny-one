cmake_minimum_required(VERSION 3.22)
project(PocketDrum246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(drum246_render_tests Drum246RenderTests.cpp)
target_include_directories(drum246_render_tests PRIVATE ../../Source)
add_test(NAME drum246_render COMMAND drum246_render_tests "${CMAKE_CURRENT_SOURCE_DIR}/vectors")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 add_executable(drum246_integration_tests ../../Tests/Drum246IntegrationTests.cpp)
 target_include_directories(drum246_integration_tests PRIVATE ../../Source)
 add_test(NAME drum246_integration COMMAND drum246_integration_tests)
endif()
