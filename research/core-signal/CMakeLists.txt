cmake_minimum_required(VERSION 3.22)
project(PocketCoreSignal246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(core_signal246_tests CoreSignal246Tests.cpp)
target_include_directories(core_signal246_tests PRIVATE ../../Source)
add_test(NAME core_signal246 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RunCoreSignal246Tests.py" "$<TARGET_FILE:core_signal246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/vectors")
add_test(NAME core_signal246_reference_inventory COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/TestReferenceInventory.py")
