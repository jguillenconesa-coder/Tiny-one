cmake_minimum_required(VERSION 3.22)
project(PocketCrank246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(crank246_tests ../../Tests/Crank246Tests.cpp)
target_include_directories(crank246_tests PRIVATE ../../Source)
if(BUILD_TESTING)
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_test(NAME crank246_original COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../Tests/RunCrank246Tests.py" "$<TARGET_FILE:crank246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/../../Tests/ReferenceVectors")
endif()
add_executable(crank246_integration_tests ../../Tests/Crank246IntegrationTests.cpp)
target_include_directories(crank246_integration_tests PRIVATE ../../Source)
add_test(NAME crank246_integration COMMAND crank246_integration_tests)
