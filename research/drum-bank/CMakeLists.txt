cmake_minimum_required(VERSION 3.22)
project(PocketDrumBank246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(drum_bank246_tests DrumBank246Tests.cpp)
target_include_directories(drum_bank246_tests PRIVATE ../../Source)
add_test(NAME drum_bank246 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RunDrumBank246Tests.py" "$<TARGET_FILE:drum_bank246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/vectors")
