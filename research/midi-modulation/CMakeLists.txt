cmake_minimum_required(VERSION 3.22)
project(PocketMidi246 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(midi246_tests ../../Tests/Midi246Tests.cpp)
target_include_directories(midi246_tests PRIVATE ../../Source)
if(BUILD_TESTING)
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_test(NAME midi246_original COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../Tests/RunMidi246Tests.py" "$<TARGET_FILE:midi246_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/../../Tests/ReferenceVectors")
endif()
add_executable(midi246_integration_tests ../../Tests/Midi246IntegrationTests.cpp)
target_include_directories(midi246_integration_tests PRIVATE ../../Source)
add_test(NAME midi246_integration COMMAND midi246_integration_tests)
